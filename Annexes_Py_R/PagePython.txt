Comandos para el intérprete de Python (IDLE por ejemplo)

 
# -------------------------------------
importar os # os modulo
os.chdir ("c: / ADT_py") # designa la carpeta que contiene el programa y los datos /
chemin = "SOTU_40_08.txt" # nombre del archivo de texto (la misma carpeta)
    # (discursos: ESTADO DE LA UNIÓN desde 1940 hasta 2008)
importar concord_E # archivo de programa: concord_E.py (misma carpeta)
desde concord_E import * # call de funciones en concord_E.py

conco (chemin, cible) # ejecutando la función conco incluida en el archivo: concord_E.py
